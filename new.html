document.addEventListener('DOMContentLoaded', () => { // --- Configuration --- const API_BASE_URL = 'http://127.0.0.1:5000'; // CHANGE IF YOUR FLASK APP IS ELSEWHERE const CALIBRATE_ENDPOINT = ${API_BASE_URL}/calibrate; const MEASURE_ENDPOINT = ${API_BASE_URL}/measure; // --- DOM Elements (IDs match the updated HTML) --- const calibrateFileFrontInput = document.getElementById('calibrateFileFront'); const calibratePreviewFront = document.getElementById('calibratePreviewFront'); const calibrateFileSideInput = document.getElementById('calibrateFileSide'); // Matches new HTML ID const calibratePreviewSide = document.getElementById('calibratePreviewSide'); // Matches new HTML ID const calibrateButton = document.getElementById('calibrateButton'); const calibrateResultDiv = document.getElementById('calibrateResult'); // Targets the result div const measureFileFrontInput = document.getElementById('measureFileFront'); const measurePreviewFront = document.getElementById('measurePreviewFront'); const measureFileSideInput = document.getElementById('measureFileSide'); const measurePreviewSide = document.getElementById('measurePreviewSide'); const measureButton = document.getElementById('measureButton'); const measureResultDiv = document.getElementById('measureResult'); // Targets the result div // MODIFIED: Matches the updated HTML ID for the overlay container const loadingIndicator = document.getElementById('loadingIndicator'); // --- Helper Functions --- function readFileAsBase64(file) { return new Promise((resolve, reject) => { if (!file) { resolve(null); return; } const reader = new FileReader(); reader.onload = () => { const base64String = reader.result.split(',')[1]; resolve(base64String); }; reader.onerror = (error) => reject(error); reader.readAsDataURL(file); }); } function updatePreview(fileInput, previewElement) { if (!fileInput || !previewElement) { console.warn("Preview update skipped: Input or Preview element not found."); return; } const placeholder = previewElement.nextElementSibling; // Assumes placeholder is sibling const file = fileInput.files[0]; if (file) { const reader = new FileReader(); reader.onload = function(e) { previewElement.src = e.target.result; previewElement.style.display = 'block'; // Show the image if (placeholder) placeholder.style.display = 'none'; // Hide placeholder } reader.readAsDataURL(file); } else { previewElement.src = '#'; previewElement.style.display = 'none'; // Hide image if (placeholder) placeholder.style.display = 'block'; // Show placeholder } } function showLoading(isLoading) { if (loadingIndicator) { // Use 'flex' display if the overlay uses flexbox for centering loadingIndicator.style.display = isLoading ? 'flex' : 'none'; } if (calibrateButton) calibrateButton.disabled = isLoading; if (measureButton) measureButton.disabled = isLoading; } // --- UPDATED: displayResult uses class names from HTML --- function displayResult(divElement, message, isSuccess) { if (!divElement) { console.error("Cannot display result: Target div element not found.", divElement); // Try to alert the user if the div is missing alert(Error displaying result: ${message}); return; } // Set message using innerHTML for line breaks divElement.innerHTML = message.replace(/\n/g, '
'); // Reset classes to only the base class divElement.className = 'result-box mt-3'; // Ensure base classes are always present // Add success or error class if applicable if (isSuccess === true) { divElement.classList.add('success'); } else if (isSuccess === false) { divElement.classList.add('error'); } } function formatFactors(factors) { let factorString = 'Calibration Factors (pixels/cm):\n'; // Use strong tag if (!factors || Object.keys(factors).length === 0) { return factorString + 'N/A'; } for (const [key, value] of Object.entries(factors)) { // Make keys more readable if needed const readableKey = key.replace('_', ' ').replace(/\b\w/g, l => l.toUpperCase()); factorString += - ${readableKey}: ${value !== null ? value.toFixed(4) : 'N/A'}\n; } return factorString; } function formatMeasurements(measurements) { let measurementString = 'Measurements (cm):\n'; // Use strong tag if (!measurements || Object.keys(measurements).length === 0) { return measurementString + 'N/A'; } for (const [key, value] of Object.entries(measurements)) { measurementString += - ${key}: ${value !== null ? value : 'N/A'}\n; } return measurementString; } // --- Event Listeners (Check if elements exist) --- if (calibrateFileFrontInput) { calibrateFileFrontInput.addEventListener('change', () => { updatePreview(calibrateFileFrontInput, calibratePreviewFront); }); } else { console.error("#calibrateFileFront not found"); } if (calibrateFileSideInput) { calibrateFileSideInput.addEventListener('change', () => { updatePreview(calibrateFileSideInput, calibratePreviewSide); }); } else { console.error("#calibrateFileSide not found"); } if (measureFileFrontInput) { measureFileFrontInput.addEventListener('change', () => { updatePreview(measureFileFrontInput, measurePreviewFront); }); } else { console.error("#measureFileFront not found"); } if (measureFileSideInput) { measureFileSideInput.addEventListener('change', () => { updatePreview(measureFileSideInput, measurePreviewSide); }); } else { console.error("#measureFileSide not found"); } // --- Calibration Button Click --- if (calibrateButton) { calibrateButton.addEventListener('click', async () => { // Ensure input elements were found before accessing files if (!calibrateFileFrontInput || !calibrateFileSideInput) { displayResult(calibrateResultDiv, "Error: Calibration file input elements not found in HTML.", false); return; } const fileFront = calibrateFileFrontInput.files[0]; const fileSide = calibrateFileSideInput.files[0]; if (!fileFront || !fileSide) { displayResult(calibrateResultDiv, 'Please select BOTH front and side calibration images.', false); return; } displayResult(calibrateResultDiv, 'Calibrating (multi-factor)...', null); showLoading(true); try { const [base64CalFront, base64CalSide] = await Promise.all([ readFileAsBase64(fileFront), readFileAsBase64(fileSide) ]); if (!base64CalFront || !base64CalSide) { throw new Error("Failed to read one or both calibration files."); } const response = await fetch(CALIBRATE_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ image_calibrate_front: base64CalFront, image_calibrate_side: base64CalSide }), }); const result = await response.json(); if (response.ok && result.status === 'calibrated') { let successMsg = 'Calibration successful!\n\n'; successMsg += formatFactors(result.calculated_factors); displayResult(calibrateResultDiv, successMsg, true); } else { // Display backend error or generic message const errorMsg = result?.error || response.statusText || "Unknown calibration error"; displayResult(calibrateResultDiv, Calibration failed: ${errorMsg}, false); } } catch (error) { console.error("Calibration fetch error:", error); displayResult(calibrateResultDiv, Network or server error during calibration: ${error.message}, false); } finally { showLoading(false); } }); } else { console.error("#calibrateButton not found"); } // --- Measurement Button Click --- if (measureButton) { measureButton.addEventListener('click', async () => { // Ensure input elements were found if (!measureFileFrontInput || !measureFileSideInput) { displayResult(measureResultDiv, "Error: Measurement file input elements not found in HTML.", false); return; } const fileFront = measureFileFrontInput.files[0]; const fileSide = measureFileSideInput.files[0]; if (!fileFront || !fileSide) { displayResult(measureResultDiv, 'Please select both front and side images for measurement.', false); return; } displayResult(measureResultDiv, 'Measuring...', null); showLoading(true); try { const [base64Front, base64Side] = await Promise.all([ readFileAsBase64(fileFront), readFileAsBase64(fileSide) ]); if (!base64Front || !base64Side) { throw new Error("Failed to read one or both measurement files."); } const response = await fetch(MEASURE_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ image_front: base64Front, image_side: base64Side }), }); const result = await response.json(); if (response.ok && result.status === 'success') { let resultString = 'Measurement successful!\n\n'; // Format measurements first resultString += formatMeasurements(result.measurements_cm); // Then format factors used resultString += \n${formatFactors(result.calibration_factors_used)}; displayResult(measureResultDiv, resultString, true); } else { const errorMsg = result?.error || response.statusText || "Unknown measurement error"; displayResult(measureResultDiv, Measurement failed: ${errorMsg}, false); } } catch (error) { console.error("Measurement fetch error:", error); displayResult(measureResultDiv, Network or server error during measurement: ${error.message}, false); } finally { showLoading(false); } }); } else { console.error("#measureButton not found"); } // Final check for essential elements after setting up listeners if (!calibrateResultDiv || !measureResultDiv || !loadingIndicator) { console.error("Initialization Error: One or more result/loading elements were not found. Check HTML IDs."); alert("Page Initialization Error: Some elements are missing. Please check the console (F12)."); } }); // End DOMContentLoaded
